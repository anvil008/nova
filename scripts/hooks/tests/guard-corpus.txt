# build-guard probe corpus — one probe per line, `expected|command`.
# expected is `allow` or `deny`; everything after the first `|` is the raw shell command.
# test_hooks.sh runs every probe through build-guard for each default payload shape and
# asserts the verdict. An expectation may name one shape instead, as `expected@shape`:
# `claude` / `codex` for a harness's own top-level session, `claude-agent` / `codex-agent`
# for a payload naming the subagent that made the call, `agy` for Antigravity's. The three
# `-eval` shapes (`claude-eval`, `claude-agent-eval`, `agy-eval`) send the same payloads from a
# separate repository carrying .workcell/eval-mode.json at its root; `claude-nested` sends them
# from an ordinary repository whose parent carries one and `claude-stray` from a plain directory
# that does. Every unscoped probe runs outside all of them, which is what proves eval mode changes
# nothing anywhere else. Keep the deny set to unambiguous policy violations and the allow set to
# the false positives we have actually hit.

# --- eval mode belongs to the toplevel alone: no upward walk, no unresolvable directory
# `claude-nested` runs in an ordinary repository whose PARENT carries a marker; `claude-stray` in
# a plain directory that carries one. Neither is a task repository, so neither is relaxed — and
# `gh` staying allowed there is what proves the guard did not read eval mode into it.

deny@claude-nested|git commit -m x
deny@claude-nested|git merge feat
deny@claude-nested|jj git push

deny@claude-stray|git push origin main
deny@claude-stray|jj git push --all
allow@claude-stray|gh pr merge 12 --merge

# --- deny: writing the eval-mode marker, in eval mode or out of it -------------------
# Entering eval mode is a human running bootstrap-eval.sh. Denied everywhere, so a marked
# repository cannot be used to mark another one either.
deny|echo {} > /task/.workcell/eval-mode.json
deny|echo {} >/task/.workcell/eval-mode.json
deny|printf '{}' >> .workcell/eval-mode.json
deny|mkdir -p .workcell && echo {} > .workcell/eval-mode.json
deny|cat seed.json | tee /task/.workcell/eval-mode.json
deny|cp /seed.json /task/.workcell/eval-mode.json
deny|mv /seed.json .workcell/eval-mode.json
deny|install -m 644 /seed.json /task/.workcell/eval-mode.json
deny|ln -s /seed.json /task/.workcell/eval-mode.json
deny|bash -c "echo {} > /task/.workcell/eval-mode.json"
deny@claude-eval|echo {} > /other/.workcell/eval-mode.json
deny@agy-eval|cp /seed.json /other/.workcell/eval-mode.json

# --- allow: reading the marker, and paths that only resemble it ----------------------
